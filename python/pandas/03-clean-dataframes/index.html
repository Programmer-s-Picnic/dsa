Before we begin

What makes data dirty?

Real tables can contain missing marks, duplicates, inconsistent city spellings, spaces, numbers stored as text, impossible scores and incomplete records. Cleaning means applying explicit rules and recording what changed—not simply removing anything unusual.

Remove only exact duplicate rows and rows without the required name. A repeated student ID with conflicting details must be reviewed rather than silently dropped.

Step 6 / 11

Validate marks using business rules

Student marks must lie between 0 and 100. Invalid entries must be flagged for investigation. This beginner example sets impossible scores to missing rather than guessing a correction.

Python · pandas

students = pd.DataFrame([
    {"StudentID": 101, "Name": " Asha ", "City": " Varanasi ", "Marks": "88", "Attendance": 92},
    {"StudentID": 102, "Name": "Ravi", "City": "ranchi", "Marks": "42", "Attendance": 75},
    {"StudentID": 103, "Name": "Meera", "City": "VARANASI", "Marks": None, "Attendance": 80},
    {"StudentID": 103, "Name": "Meera", "City": "VARANASI", "Marks": None, "Attendance": 80},
    {"StudentID": 104, "Name": " Kabir", "City": "Patna", "Marks": "108", "Attendance": 50},
    {"StudentID": 105, "Name": "Nina", "City": None, "Marks": "-4", "Attendance": None},
    {"StudentID": 106, "Name": None, "City": "Ranchi", "Marks": "absent", "Attendance": 65},
    {"StudentID": 107, "Name": "Rahul", "City": "varanasi", "Marks": "91", "Attendance": 95},
])

students["Marks"] = pd.to_numeric(students["Marks"], errors="coerce")
invalid = students["Marks"].notna() & ~students["Marks"].between(0, 100)
print("Invalid values found:")
print(students.loc[invalid, ["StudentID", "Marks"]])
students.loc[invalid, "Marks"] = float("nan")
print(students[["Name", "Marks"]])

Remember: Never replace a score of 108 with 100 automatically unless a documented policy authorises that choice.

Step 7 / 11

Decide which missing values to fill

Different columns require different decisions: keep unknown marks missing, use "Unknown" for missing city, and optionally fill missing attendance with the observed median.

Python · pandas
import pandas as pd

students = pd.DataFrame([
    {"StudentID": 101, "Name": " Asha ", "City": " Varanasi ", "Marks": "88", "Attendance": 92},
    {"StudentID": 102, "Name": "Ravi", "City": "ranchi", "Marks": "42", "Attendance": 75},
    {"StudentID": 103, "Name": "Meera", "City": "VARANASI", "Marks": None, "Attendance": 80},
    {"StudentID": 103, "Name": "Meera", "City": "VARANASI", "Marks": None, "Attendance": 80},
    {"StudentID": 104, "Name": " Kabir", "City": "Patna", "Marks": "108", "Attendance": 50},
    {"StudentID": 105, "Name": "Nina", "City": None, "Marks": "-4", "Attendance": None},
    {"StudentID": 106, "Name": None, "City": "Ranchi", "Marks": "absent", "Attendance": 65},
    {"StudentID": 107, "Name": "Rahul", "City": "varanasi", "Marks": "91", "Attendance": 95},
])

students["Marks"] = pd.to_numeric(students["Marks"], errors="coerce")
invalid = students["Marks"].notna() & ~students["Marks"].between(0, 100)
students.loc[invalid, "Marks"] = float("nan")
students["City"] = students["City"].astype("string").str.strip().replace("", pd.NA).fillna("Unknown")
median = students["Attendance"].median()
if pd.notna(median):
    students["Attendance"] = students["Attendance"].fillna(median)
students["MarksMissing"] = students["Marks"].isna()
print(students[["Name", "City", "Marks", "MarksMissing", "Attendance"]])

Remember: Imputation is a modelling choice. Record it and check whether median filling is appropriate for your problem.

Remember: A small class register is not suitable for strong outlier conclusions. Flag, investigate, and retain records until you have evidence.

Complete mini-project

Clean the entire class register with one function

This fully runnable Python program creates eight records directly, works on a copy, reports every cleaning decision, validates the result and outputs six rows. It does not invent missing exam marks.

Python · pandas
"""Pandas 0 to Infinity, Level 3: DataFrame cleaning without external files."""

import pandas as pd


def make_messy_students():
    """Create an intentionally messy but entirely in-memory class register."""
    return pd.DataFrame([
        {"StudentID": 101, "Name": " Asha ", "City": " Varanasi ", "Marks": "88", "Attendance": 92},
        {"StudentID": 102, "Name": "Ravi", "City": "ranchi", "Marks": "42", "Attendance": 75},
        {"StudentID": 103, "Name": "Meera", "City": "VARANASI", "Marks": None, "Attendance": 80},
        {"StudentID": 103, "Name": "Meera", "City": "VARANASI", "Marks": None, "Attendance": 80},
        {"StudentID": 104, "Name": " Kabir", "City": "Patna", "Marks": "108", "Attendance": 50},
        {"StudentID": 105, "Name": "Nina", "City": None, "Marks": "-4", "Attendance": None},
        {"StudentID": 106, "Name": None, "City": "Ranchi", "Marks": "absent", "Attendance": 65},
        {"StudentID": 107, "Name": "Rahul", "City": "varanasi", "Marks": "91", "Attendance": 95},
    ])


def clean_students(raw):
    """Return a new cleaned table and an explicit audit; leave raw untouched."""
    df = raw.copy(deep=True)
    audit = {"input_rows": len(df), "missing_before": df.isna().sum().to_dict()}

    for column in ("Name", "City"):
        df[column] = df[column].astype("string").str.strip().replace("", pd.NA).str.title()

    audit["exact_duplicate_rows_removed"] = int(df.duplicated().sum())
    df = df.drop_duplicates().copy()

    supplied_marks = df["Marks"].notna()
    df["Marks"] = pd.to_numeric(df["Marks"], errors="coerce")
    audit["unparseable_marks"] = int((supplied_marks & df["Marks"].isna()).sum())

    bad_marks = df["Marks"].notna() & ~df["Marks"].between(0, 100)
    audit["marks_outside_0_100"] = int(bad_marks.sum())
    df.loc[bad_marks, "Marks"] = float("nan")

    df["Attendance"] = pd.to_numeric(df["Attendance"], errors="coerce")
    bad_attendance = df["Attendance"].notna() & ~df["Attendance"].between(0, 100)
    audit["invalid_attendance"] = int(bad_attendance.sum())
    df.loc[bad_attendance, "Attendance"] = float("nan")

    audit["rows_without_name_removed"] = int(df["Name"].isna().sum())
    df = df.dropna(subset=["Name"]).copy()
    audit["cities_marked_unknown"] = int(df["City"].isna().sum())
    df["City"] = df["City"].fillna("Unknown")

    attendance_median = df["Attendance"].median()
    audit["attendance_filled_with_median"] = (
        int(df["Attendance"].isna().sum()) if pd.notna(attendance_median) else 0
    )
    if pd.notna(attendance_median):
        df["Attendance"] = df["Attendance"].fillna(attendance_median)
    df["MarksMissing"] = df["Marks"].isna()
    df = df.reset_index(drop=True)

    assert df["StudentID"].is_unique, "Review conflicting StudentID records manually"
    assert df["Name"].notna().all(), "Name is required"
    assert df["Marks"].dropna().between(0, 100).all(), "Invalid marks remain"
    assert df["Attendance"].dropna().between(0, 100).all(), "Invalid attendance remains"

    audit["output_rows"] = len(df)
    audit["unresolved_marks"] = int(df["MarksMissing"].sum())
    return df, audit


if __name__ == "__main__":
    original = make_messy_students()
    cleaned, report = clean_students(original)
    print("Original records:\n", original.to_string(index=False))
    print("\nCleaned records:\n", cleaned.to_string(index=False))
    print("\nCleaning audit:")
    for key, value in report.items():
        print(f"- {key}: {value}")

Construct a DataFrame with Item, Price and Stock. Introduce one repeated row, one blank item, the text "free" in Price, and a negative stock value. Remove exact duplicates, mark unparseable prices as missing, flag negative stock and print an audit.

Python · pandas
import pandas as pd

shop = pd.DataFrame({
    "Item": [" Pen ", "Notebook", "Notebook", "", "Eraser"],
    "Price": ["10", "50", "50", "5", "free"],
    "Stock": [30, 12, 12, 4, -3]
})
raw = shop.copy(deep=True)
shop["Item"] = shop["Item"].astype("string").str.strip().replace("", pd.NA)
duplicates = int(shop.duplicated().sum())
shop = shop.drop_duplicates().dropna(subset=["Item"]).copy()
shop["Price"] = pd.to_numeric(shop["Price"], errors="coerce")
bad_stock = shop["Stock"] < 0
shop.loc[bad_stock, "Stock"] = pd.NA
print(shop)
print({"raw_rows": len(raw), "duplicates": duplicates,
       "remaining_rows": len(shop), "missing_prices": int(shop["Price"].isna().sum()),
       "missing_stock": int(shop["Stock"].isna().sum())})
assert raw.shape[0] == 5
